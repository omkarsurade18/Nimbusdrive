// FIX #1 & #2: Signup never auto-logs in; fields are always cleared
function doSignup() {
  const name = document.getElementById('signupName').value.trim();
  const email = document.getElementById('signupEmail').value.trim().toLowerCase();
  const password = document.getElementById('signupPassword').value;

  if (!name || !email || !password) return showAuthError('Please fill in all fields.');
  if (password.length < 6) return showAuthError('Password must be at least 6 characters.');
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) return showAuthError('Please enter a valid email.');

  const users = getUsers();
  if (users[email]) return showAuthError('Account already exists. Please sign in.');

  users[email] = { name, email, password: btoa(password), createdAt: new Date().toISOString() };
  saveUsers(users);

  showAuthSuccess('✅ Account created successfully! Please sign in.');
  setTimeout(() => {
    switchTab('login');
    document.getElementById('loginEmail').value = email;
    document.getElementById('loginPassword').value = '';
  }, 1400);
}

function doLogin() {
  const email = document.getElementById('loginEmail').value.trim().toLowerCase();
  const password = document.getElementById('loginPassword').value;

  if (!email || !password) return showAuthError('Please fill in all fields.');

  // Try API first
  fetch(`${API}/auth/login`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ email, password })
  })
  .then(r => r.ok ? r.json() : Promise.reject())
  .then(d => {
    if (d.token) startSession(d.token, d.user);
    else showAuthError(d.error || 'Login failed');
  })
  .catch(() => {
    // Local auth fallback
    const users = getUsers();
    const user = users[email];
    if (!user) return showAuthError('No account found. Please sign up first.');
    try {
      if (atob(user.password) !== password) return showAuthError('Incorrect password.');
    } catch { return showAuthError('Incorrect password.'); }
    startSession('local-' + Date.now(), { name: user.name, email: user.email });
  });
}

// FIX #4: Precise type mappings for folder filtering
const IMG_EXTS    = ['jpg','jpeg','png','gif','webp','svg','bmp','avif','heic','tiff'];
const VIDEO_EXTS  = ['mp4','mov','avi','mkv','webm','m4v','flv','wmv','3gp'];
const AUDIO_EXTS  = ['mp3','wav','flac','aac','ogg','m4a','wma','opus'];
const DOC_EXTS    = ['pdf','doc','docx','txt','md','rtf','odt','pages'];
const SHEET_EXTS  = ['xls','xlsx','csv','numbers','ods'];
const ZIP_EXTS    = ['zip','rar','7z','tar','gz','bz2','xz'];

function getFileType(name) {
  const ext = (name || '').split('.').pop().toLowerCase();
  if (IMG_EXTS.includes(ext))    return 'img';
  if (VIDEO_EXTS.includes(ext))  return 'video';
  if (AUDIO_EXTS.includes(ext))  return 'audio';
  if (DOC_EXTS.includes(ext))    return 'doc';
  if (SHEET_EXTS.includes(ext))  return 'sheet';
  if (ZIP_EXTS.includes(ext))    return 'zip';
  return 'default';
}

function getFileEmoji(name) {
  const map = { img:'🖼️', video:'🎬', audio:'🎵', doc:'📝', sheet:'📊', zip:'🗜️', default:'📁' };
  return map[getFileType(name)] || '📁';
}

function formatSize(b) {
  if (!b || b === 0) return '0 B';
  if (b < 1024) return b + ' B';
  if (b < 1048576) return (b / 1024).toFixed(1) + ' KB';
  if (b < 1073741824) return (b / 1048576).toFixed(1) + ' MB';
  return (b / 1073741824).toFixed(2) + ' GB';
}

function timeAgo(d) {
  if (!d) return '—';
  const diff = Date.now() - new Date(d), m = Math.floor(diff / 60000);
  if (m < 1) return 'Just now';
  if (m < 60) return `${m}m ago`;
  const h = Math.floor(m / 60);
  if (h < 24) return `${h}h ago`;
  const dy = Math.floor(h / 24);
  if (dy < 30) return `${dy}d ago`;
  return new Date(d).toLocaleDateString();
}

async function loadFiles() {
  try {
    const r = await fetch(`${API}/files`, { headers: { Authorization: `Bearer ${authToken}` } });
    if (r.status === 401) { doLogout(); return; }
    const d = await r.json();
    if (d.files && d.files.length > 0) {
      allFiles = d.files.map(f => ({
        fileId: f.fileId, name: f.fileName, size: f.fileSize, uploadedAt: f.uploadedAt
      }));
      saveUserFiles();
      updateStats(); renderFiles(); updateBadges();
    }
  } catch {
    // Server not running — local mode is fine
  }
}

function sortFiles(a, s) {
  a.sort((x, y) => {
    if (s === 'date-desc') return new Date(y.uploadedAt||0) - new Date(x.uploadedAt||0);
    if (s === 'date-asc')  return new Date(x.uploadedAt||0) - new Date(y.uploadedAt||0);
    if (s === 'name-asc')  return (x.name||'').localeCompare(y.name||'');
    if (s === 'name-desc') return (y.name||'').localeCompare(x.name||'');
    if (s === 'size-desc') return (y.size||0) - (x.size||0);
    if (s === 'size-asc')  return (x.size||0) - (y.size||0);
    return 0;
  });
}

function uploadFile(file) {
  return new Promise(resolve => {
    const wrap = document.getElementById('progressWrap');
    const fill = document.getElementById('progressFill');
    const pctEl = document.getElementById('progressPct');
    const nameEl = document.getElementById('progressName');

    wrap.style.display = 'block';
    nameEl.textContent = file.name;
    fill.style.width = '0%';
    pctEl.textContent = '0%';

    let progress = 0;
    const sim = setInterval(() => {
      progress += Math.random() * 15 + 5;
      if (progress >= 92) clearInterval(sim);
      const p = Math.min(progress, 92);
      fill.style.width = p + '%';
      pctEl.textContent = Math.round(p) + '%';
    }, 120);

    const formData = new FormData();
    formData.append('file', file);

    const xhr = new XMLHttpRequest();
    xhr.open('POST', `${API}/upload`);
    xhr.setRequestHeader('Authorization', `Bearer ${authToken}`);

    xhr.upload.onprogress = e => {
      if (e.lengthComputable) {
        clearInterval(sim);
        const p = Math.round((e.loaded / e.total) * 100);
        fill.style.width = p + '%';
        pctEl.textContent = p + '%';
      }
    };

    const finish = (id) => {
      clearInterval(sim);
      fill.style.width = '100%';
      pctEl.textContent = '100%';
      const newFile = {
        fileId: id || ('f_' + Date.now() + '_' + Math.random().toString(36).slice(2)),
        name: file.name,
        size: file.size,
        uploadedAt: new Date().toISOString()
      };
      allFiles.unshift(newFile);
      saveUserFiles();
      updateStats(); renderFiles(); updateBadges();
      showToast(`✅ ${file.name} uploaded!`, 'success');
      setTimeout(() => { wrap.style.display = 'none'; resolve(); }, 700);
    };

    xhr.onload = () => {
      try { const d = JSON.parse(xhr.responseText); finish(d.fileId); }
      catch { finish(); }
    };
    xhr.onerror = () => finish();
    xhr.send(formData);
  });
}

async function downloadFile(fileId) {
  try {
    const r = await fetch(`${API}/download/${fileId}`, { headers: { Authorization: `Bearer ${authToken}` } });
    const d = await r.json();
    if (d.downloadUrl) { window.open(d.downloadUrl, '_blank'); showToast('⬇ Download started', 'info'); }
    else showToast('⬇ Download ready (demo mode)', 'info');
  } catch { showToast('⬇ Preparing download…', 'info'); }
}

async function deleteFile(fileId, name) {
  if (!confirm(`Move "${name}" to trash?`)) return;
  const f = allFiles.find(x => x.fileId === fileId);
  if (f) { f.deletedAt = new Date().toISOString(); trashFiles.push({ ...f }); }
  allFiles = allFiles.filter(x => x.fileId !== fileId);
  try { await fetch(`${API}/delete/${fileId}`, { method: 'DELETE', headers: { Authorization: `Bearer ${authToken}` } }); } catch {}
  saveUserFiles(); updateStats(); renderFiles(); updateBadges();
  showToast('🗑 Moved to trash', 'warning');
}

function getEffectiveFolders() {
  return (allFolders && allFolders.length) ? allFolders : DEFAULT_FOLDERS;
}

function getFolderFiles(folder) {
  if (!folder) return allFiles;
  if (folder.filterType) {
    // Built-in typed folders: strict filtering by file type
    return allFiles.filter(f => getFileType(f.name) === folder.filterType);
  }
  // Custom folders: files manually assigned
  return allFiles.filter(f => f.folderId === folder.id);
}

function renderFolders() {
  const grid = document.getElementById('folderGrid');
  const folders = getEffectiveFolders();
  grid.innerHTML = folders.map((f, i) => {
    const count = getFolderFiles(f).length;
    return `<div class="folder-card" style="animation-delay:${i * 0.06}s" onclick="openFolder('${esc(f.id)}')">
      <div class="folder-icon">${f.icon}</div>
      <div class="folder-name">${esc(f.name)}</div>
      <div class="folder-count">${count} file${count !== 1 ? 's' : ''}</div>
    </div>`;
  }).join('') + `<div class="folder-card folder-add" onclick="openNewFolderModal()"><span>➕</span><p>New Folder</p></div>`;

  document.getElementById('folderFileList').innerHTML = emptyStateHTML('📁', 'Select a Folder', 'Click a folder above to view its files');
}

function renderStorageDash() {
  const total = allFiles.reduce((s, f) => s + (f.size || 0), 0);
  const pct = Math.min((total / TOTAL) * 100, 100);
  const free = Math.max(TOTAL - total, 0);

  const [uv, uu] = formatSize(total).split(' ');
  const [fv, fu] = formatSize(free).split(' ');
  document.getElementById('dashUsed').innerHTML = `${uv} <span>${uu||''}</span>`;
  document.getElementById('dashFree').innerHTML = `${fv} <span>${fu||''}</span>`;
  document.getElementById('dashFiles').textContent = allFiles.length;
  document.getElementById('dashPct').textContent = pct.toFixed(2) + '%';
  setTimeout(() => { document.getElementById('dashFill').style.width = pct + '%'; }, 50);

  setTimeout(() => {
    const arc = document.getElementById('donutArc');
    if (arc) arc.style.strokeDashoffset = 239 * (1 - pct / 100);
  }, 100);
  document.getElementById('donutUsedLabel').textContent = `Used: ${formatSize(total)}`;
  document.getElementById('donutFreeLabel').textContent = `Free: ${formatSize(free)}`;

  const types = {};
  allFiles.forEach(f => {
    const t = getFileType(f.name);
    types[t] = (types[t] || 0) + (f.size || 0);
  });
  const typeTotalBytes = Object.values(types).reduce((a, b) => a + b, 1);
  const colors = { img:'#10b981', video:'#f59e0b', audio:'#a5b4fc', doc:'#06b6d4', sheet:'#34d399', zip:'#f97316', default:'#8b5cf6' };
  const labels = { img:'Images', video:'Videos', audio:'Audio', doc:'Docs', sheet:'Sheets', zip:'Zips', default:'Other' };

  const typeHTML = Object.entries(types).filter(([,v]) => v > 0)
    .sort((a,b) => b[1]-a[1])
    .map(([k, v]) => `<div class="file-type-row">
      <span class="ft-label" style="color:${colors[k]||'#8b5cf6'}">${labels[k]||k}</span>
      <div class="ft-track"><div class="ft-fill" style="width:0%;background:${colors[k]||'#8b5cf6'}" data-pct="${(v/typeTotalBytes*100).toFixed(0)}"></div></div>
      <span class="ft-pct">${(v/typeTotalBytes*100).toFixed(0)}%</span>
    </div>`).join('') || '<div style="color:var(--muted);font-size:0.83rem">No files uploaded yet</div>';
  document.getElementById('typeBreakdown').innerHTML = typeHTML;
  setTimeout(() => { document.querySelectorAll('.ft-fill').forEach(el => { el.style.width = (el.dataset.pct||0) + '%'; }); }, 100);

  const acts = allFiles.slice(0, 6).map(f => ({ text: `Uploaded ${f.name}`, time: timeAgo(f.uploadedAt) }));
  document.getElementById('activityList').innerHTML = acts.length
    ? acts.map(a => `<div class="activity-item"><div class="activity-dot" style="background:var(--accent)"></div><span class="activity-text">${esc(a.text)}</span><span class="activity-time">${a.time}</span></div>`).join('')
    : '<div style="color:var(--muted);font-size:0.83rem">No recent activity</div>';

// ════════════════════════════════════════
function openTaskModal(fileId) {
  currentTaskFileId = fileId;
  const f = allFiles.find(x => x.fileId === fileId);
  document.getElementById('taskTitle').value = f?.taskTitle || '';
  document.getElementById('taskAssignee').value = f?.taskAssignee || '';
  document.getElementById('taskDue').value = f?.taskDue || '';
  document.getElementById('taskPriority').value = f?.taskPriority || 'Medium';
  document.getElementById('taskNotes').value = f?.taskNotes || '';
  document.getElementById('taskModal').classList.add('open');
}
function closeTaskModal() { document.getElementById('taskModal').classList.remove('open'); }
function saveTask() {
  if (!currentTaskFileId) return;
  const f = allFiles.find(x => x.fileId === currentTaskFileId);
  if (f) {
    const title = document.getElementById('taskTitle').value.trim();
    f.task = title || 'Task';
    f.taskTitle = title;
    f.taskAssignee = document.getElementById('taskAssignee').value;
    f.taskDue = document.getElementById('taskDue').value;
    f.taskPriority = document.getElementById('taskPriority').value;
    f.taskNotes = document.getElementById('taskNotes').value;
    saveUserFiles(); renderFiles();
    showToast('✅ Task assigned!', 'success');
  }
  closeTaskModal();
}

// VOICE SEARCH — FIX #6: Real Web Speech API
// ════════════════════════════════════════
function startVoiceSearch() {
  const SpeechRec = window.SpeechRecognition || window.webkitSpeechRecognition;
  if (!SpeechRec) {
    showToast('Voice search not supported in this browser', 'error');
    return;
  }

  // Stop any ongoing recognition
  if (speechRecognition) {
    try { speechRecognition.stop(); } catch(e){}
    speechRecognition = null;
  }

  const btn = document.getElementById('voiceSearchBtn');
  const bar = document.getElementById('listeningBar');
  const input = document.getElementById('globalSearch');

  // Check if already listening — toggle off
  if (btn.classList.contains('listening')) {
    btn.classList.remove('listening');
    bar.classList.remove('active');
    return;
  }

  speechRecognition = new SpeechRec();
  speechRecognition.continuous = false;
  speechRecognition.interimResults = true;
  speechRecognition.lang = 'en-US';
  speechRecognition.maxAlternatives = 1;

  btn.classList.add('listening');
  bar.classList.add('active');
  showToast('🎤 Listening…', 'info');

  speechRecognition.onresult = (event) => {
    const result = event.results[0];
    const transcript = result[0].transcript;
    input.value = transcript;
    onGlobalSearch(transcript);

    if (result.isFinal) {
      btn.classList.remove('listening');
      bar.classList.remove('active');
      showToast(`🔍 Searching: "${transcript}"`, 'info');
    }
  };

  speechRecognition.onend = () => {
    btn.classList.remove('listening');
    bar.classList.remove('active');
    speechRecognition = null;
  };

  speechRecognition.onerror = (e) => {
    btn.classList.remove('listening');
    bar.classList.remove('active');
    speechRecognition = null;
    const msgs = { 'no-speech':'No speech detected', 'not-allowed':'Microphone access denied', 'network':'Network error' };
    showToast(msgs[e.error] || 'Voice search error', 'error');
  };

  try { speechRecognition.start(); }
  catch(e) { btn.classList.remove('listening'); bar.classList.remove('active'); }
}

// Global search with autocomplete
function onGlobalSearch(q) {
  const ac = document.getElementById('voiceAutocomplete');
  if (!q || q.length < 1) { hideAutocomplete(); return; }

  const lower = q.toLowerCase();
  const matches = allFiles
    .filter(f => f.name.toLowerCase().includes(lower))
    .slice(0, 6);

  if (!matches.length) { hideAutocomplete(); return; }

  ac.innerHTML = matches.map(f =>
    `<div class="autocomplete-item" onmousedown="selectAutocomplete('${esc(f.name)}')">
      <span class="ac-icon">${getFileEmoji(f.name)}</span>
      <span>${esc(f.name)}</span>
      <span style="margin-left:auto;font-size:0.7rem;color:var(--muted)">${formatSize(f.size)}</span>
    </div>`
  ).join('');
  ac.classList.add('show');

  // Also scroll to and highlight in home file list
  if (currentPage === 'home') {
    document.getElementById('searchInput').value = q;
    renderFiles();
  }
}

function selectAutocomplete(name) {
  document.getElementById('globalSearch').value = name;
  document.getElementById('searchInput').value = name;
  hideAutocomplete();
  renderFiles();
  if (currentPage !== 'home') goPage('home');
}

function hideAutocomplete() {
  document.getElementById('voiceAutocomplete').classList.remove('show');
}

function saveProfile() {
  const profile = {
    name:    document.getElementById('pfName').value.trim(),
    display: document.getElementById('pfDisplay').value.trim(),
    email:   document.getElementById('pfEmail').value.trim(),
    mobile:  document.getElementById('pfMobile').value.trim(),
    dob:     document.getElementById('pfDob').value,
    gender:  document.getElementById('pfGender').value,
    city:    document.getElementById('pfCity').value.trim(),
    country: document.getElementById('pfCountry').value.trim(),
    bio:     document.getElementById('pfBio').value.trim(),
  };
  localStorage.setItem(userKey('profile'), JSON.stringify(profile));

  const displayName = profile.display || profile.name || currentUser.name;
  document.getElementById('userName').textContent = displayName;

  const savedAvatar = localStorage.getItem(userKey('avatar'));
  if (!savedAvatar) {
    const initials = (displayName || '?')[0].toUpperCase();
    document.getElementById('userAvatar').textContent = initials;
    document.getElementById('avatarPreview').textContent = initials;
  }

  showToast('💾 Profile saved!', 'success');
}

function handleAvatarUpload(event) {
  const file = event.target.files[0];
  if (!file) return;
  if (file.size > 2 * 1024 * 1024) { showToast('Avatar must be under 2MB', 'error'); return; }
  const reader = new FileReader();
  reader.onload = e => {
    const data = e.target.result;
    localStorage.setItem(userKey('avatar'), data);
    document.getElementById('avatarPreview').innerHTML = `<img src="${data}" alt="avatar"/>`;
    document.getElementById('userAvatar').innerHTML = `<img src="${data}" alt="avatar"/>`;
    showToast('✅ Profile photo updated!', 'success');
  };
  reader.readAsDataURL(file);
}

function changePassword() {
  const current = document.getElementById('secCurrentPwd').value;
  const newPwd = document.getElementById('secNewPwd').value;
  const confirm = document.getElementById('secConfirmPwd').value;

  if (!current || !newPwd || !confirm) { showToast('Please fill all password fields', 'error'); return; }

  const users = getUsers();
  const email = currentUser?.email;
  if (!email || !users[email]) { showToast('User not found', 'error'); return; }

  try {
    if (atob(users[email].password) !== current) { showToast('Current password is incorrect', 'error'); return; }
  } catch { showToast('Current password is incorrect', 'error'); return; }

  if (newPwd.length < 6) { showToast('New password must be at least 6 characters', 'error'); return; }
  if (newPwd !== confirm) { showToast('New passwords do not match', 'error'); return; }

  users[email].password = btoa(newPwd);
  saveUsers(users);
  document.getElementById('secCurrentPwd').value = '';
  document.getElementById('secNewPwd').value = '';
  document.getElementById('secConfirmPwd').value = '';
  showToast('✅ Password updated successfully!', 'success');
}

// TOAST SYSTEM
// ════════════════════════════════════════
function showToast(msg, type = 'info') {
  const container = document.getElementById('toast');
  const icons = { success:'✅', error:'❌', info:'ℹ️', warning:'⚠️' };
  const el = document.createElement('div');
  el.className = `toast-item toast-${type}`;
  el.innerHTML = `<span>${icons[type]||'ℹ️'}</span><span>${msg}</span>`;
  container.appendChild(el);
  setTimeout(() => {
    el.classList.add('out');
    setTimeout(() => el.remove(), 350);
  }, 3500);
}
